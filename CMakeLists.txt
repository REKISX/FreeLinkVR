cmake_minimum_required(VERSION 3.20)
project(FreeLinkVR VERSION 0.1.0 LANGUAGES CXX)
set(FREELINKVR_PACKAGE "${CMAKE_SOURCE_DIR}/build")
if(CMAKE_BINARY_DIR STREQUAL FREELINKVR_PACKAGE)
  message(FATAL_ERROR "build/ is the ready-to-run package. Use a separate CMake directory, for example ../FreeLinkVR-build.")
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "FreeLinkVR requires 64-bit Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
add_library(slvr_common INTERFACE)
target_include_directories(slvr_common INTERFACE src)
target_include_directories(slvr_common SYSTEM INTERFACE assets/vendor/openvr)
target_link_libraries(slvr_common INTERFACE d3d11 dxgi)
add_library(freelink_platform STATIC src/platform/setup.cpp src/platform/input_context.cpp)
target_include_directories(freelink_platform PUBLIC src)
target_link_libraries(freelink_platform PUBLIC shell32 advapi32 wbemuuid ole32 oleaut32)
add_library(driver_freelinkvr SHARED src/driver/driver.cpp)
target_link_libraries(driver_freelinkvr PRIVATE slvr_common)
set_target_properties(driver_freelinkvr PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${FREELINKVR_PACKAGE}/freelinkvr/bin/win64"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${FREELINKVR_PACKAGE}/freelinkvr/bin/win64"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${FREELINKVR_PACKAGE}/freelinkvr/bin/win64")
add_executable(FreeLinkVR-Stream WIN32 src/stream/main.cpp src/stream/input_diagnostics.cpp src/stream/vr_library.cpp)
target_link_libraries(FreeLinkVR-Stream PRIVATE slvr_common freelink_platform d3dcompiler dcomp ws2_32 gdi32 xinput9_1_0)
add_executable(FreeLinkVR WIN32 src/launcher/main.cpp)
target_link_libraries(FreeLinkVR PRIVATE freelink_platform comctl32 gdi32)
foreach(app FreeLinkVR FreeLinkVR-Stream)
  set_target_properties(${app} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${FREELINKVR_PACKAGE}" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${FREELINKVR_PACKAGE}" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${FREELINKVR_PACKAGE}")
endforeach()
file(COPY assets/driver/ DESTINATION "${FREELINKVR_PACKAGE}/freelinkvr")
file(COPY LICENSE THIRD_PARTY_NOTICES.md DESTINATION "${FREELINKVR_PACKAGE}")
foreach(readme README.md README.ru.md)
  file(READ "${CMAKE_SOURCE_DIR}/${readme}" package_readme)
  string(REPLACE "(assets/docs/" "(docs/" package_readme "${package_readme}")
  file(WRITE "${FREELINKVR_PACKAGE}/${readme}" "${package_readme}")
endforeach()
file(COPY Start.cmd DESTINATION "${FREELINKVR_PACKAGE}")
file(COPY src/tools/register-driver.ps1 DESTINATION "${FREELINKVR_PACKAGE}/tools")
file(COPY assets/steaminput/ DESTINATION "${FREELINKVR_PACKAGE}/steaminput")
file(COPY assets/docs/ DESTINATION "${FREELINKVR_PACKAGE}/docs")
file(COPY assets/vendor/openvr/LICENSE DESTINATION "${FREELINKVR_PACKAGE}/licenses/openvr")
enable_testing()
add_executable(slvr_tests src/tests/tests.cpp)
target_link_libraries(slvr_tests PRIVATE slvr_common freelink_platform ws2_32)
add_test(NAME protocol_math_ipc_graphics COMMAND slvr_tests)
set_tests_properties(protocol_math_ipc_graphics PROPERTIES TIMEOUT 30)
add_test(NAME viewer_hidden_window_shaders COMMAND FreeLinkVR-Stream --self-test)
add_test(NAME launcher_hidden_window COMMAND FreeLinkVR --self-test)
set_tests_properties(viewer_hidden_window_shaders PROPERTIES TIMEOUT 30)
add_executable(slvr_driver_test src/tests/driver_test.cpp)
target_link_libraries(slvr_driver_test PRIVATE slvr_common)
add_dependencies(slvr_driver_test driver_freelinkvr)
add_test(NAME driver_dll_integration COMMAND slvr_driver_test "$<TARGET_FILE:driver_freelinkvr>")
set_tests_properties(driver_dll_integration PROPERTIES TIMEOUT 30)
# Viewer and driver tests use the production IPC names. Do not run them together.
set_tests_properties(viewer_hidden_window_shaders driver_dll_integration PROPERTIES RESOURCE_LOCK slvr_production_ipc)
add_executable(FreeLinkVR-TestScene src/tests/tools/runtime_scene.cpp)
target_link_libraries(FreeLinkVR-TestScene PRIVATE slvr_common d3dcompiler)
set_target_properties(FreeLinkVR-TestScene PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/tools"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/tools")
add_executable(freelink_setup_check EXCLUDE_FROM_ALL src/tests/tools/setup_check.cpp)
target_link_libraries(freelink_setup_check PRIVATE freelink_platform)
set_target_properties(freelink_setup_check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/tools" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/tools")
add_executable(freelink_runtime_probe EXCLUDE_FROM_ALL src/tests/tools/runtime_probe.cpp)
target_link_libraries(freelink_runtime_probe PRIVATE slvr_common freelink_platform gdi32)
set_target_properties(freelink_runtime_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/tools" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/tools")
